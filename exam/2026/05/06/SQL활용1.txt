SELECT e.empno AS no, e.ename AS name, e.sal AS salary, d.dname AS d_name, d.loc AS location
FROM emp e JOIN dept d ON e.deptno = d.deptno
WHERE e.sal >= 2000
ORDER BY sal DESC;
